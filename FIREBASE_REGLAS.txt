FIRESTORE DATABASE → REGLAS

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null && request.auth.uid == "KSYoyWiYB4dChzlwVFAaWIyN0YJ3";
    }

    match /recuerdos/{recuerdoId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();

      match /respuestas/{respuestaId} {
        allow read: if true;
        allow create: if request.auth != null
          && respuestaId == "aris"
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.keys().hasOnly(["ownerUid", "autor", "corazon", "nota", "updatedAt", "leida", "leidaAt"]);
        allow update: if request.auth != null
          && respuestaId == "aris"
          && (resource.data.ownerUid == request.auth.uid || isAdmin())
          && request.resource.data.ownerUid == resource.data.ownerUid
          && request.resource.data.keys().hasOnly(["ownerUid", "autor", "corazon", "nota", "updatedAt", "leida", "leidaAt"]);
        allow delete: if isAdmin();
      }
    }

    match /configuracion/{documentoId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
